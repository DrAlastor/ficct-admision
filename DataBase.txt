DROP TABLE IF EXISTS MODULO, FUNCION, ACCION, ROL, ROL_FUNCION, USUARIO, TOKEN_RECUPERACION, 
    BITACORA, PERFIL, ADMINISTRACION, POSTULANTE, DOCENTE, FACULTAD, 
    GESTION, CARRERA, POSTULACION, POSTULACION_CARRERA, PAGO, 
    DOCUMENTOS, MATERIA, AULA, TURNO, GRUPO, HORARIO, CARGA_HORARIA, 
    INSCRIPCIONES_CUP, EVALUACIONES CASCADE;

-- 1. MÓDULOS Y FUNCIONES (SISTEMA DE PERMISOS DINÁMICOS)
CREATE TABLE MODULO (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL,
    DESCRIPCION TEXT
);

CREATE TABLE FUNCION (
    ID SERIAL PRIMARY KEY,
    MODULO_ID INT REFERENCES MODULO(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    NOMBRE VARCHAR(100) NOT NULL,
    PERMISO VARCHAR(100) NOT NULL,
    DESCRIPCION TEXT
);

CREATE TABLE ROL (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL,
    DESCRIPCION TEXT
);

-- NUEVO
CREATE TABLE ACCION(
    ID_ACCION INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE ROL_FUNCION (
    ROL_ID INT REFERENCES ROL(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    FUNCION_ID INT REFERENCES FUNCION(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    ID_ACCION INT REFERENCES ACCION (ID_ACCION) ON DELETE RESTRICT,
    DESCRIPCION VARCHAR(50) NOT NULL,
    PRIMARY KEY (ROL_ID, FUNCION_ID)
);

-- 2. AUTENTICACIÓN Y SEGURIDAD
CREATE TABLE USUARIO (
    ID SERIAL PRIMARY KEY,
    ROL_ID INT REFERENCES ROL(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    CODIGO_INICIO VARCHAR(150) UNIQUE NOT NULL, 
    PASSWORD VARCHAR(255) NOT NULL,
    ESTADO VARCHAR(20) DEFAULT 'Activo'
);

CREATE TABLE TOKEN_RECUPERACION (
    ID SERIAL PRIMARY KEY,
    USUARIO_ID INT REFERENCES USUARIO(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    TOKEN VARCHAR(255) NOT NULL,
    FECHA_CREACION TIMESTAMP NOT NULL,
    FECHA_EXPIRACION TIMESTAMP NOT NULL,
    USADO BOOLEAN DEFAULT FALSE
);

CREATE TABLE BITACORA (
    ID SERIAL PRIMARY KEY,
    USUARIO_ID INT REFERENCES USUARIO(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    ACCION VARCHAR(255) NOT NULL,
    FECHA DATE NOT NULL,
    HORA TIME NOT NULL,
    IP VARCHAR(45)
);

-- 3. PERFIL Y SUBCLASES (ACTORES)
CREATE TABLE PERFIL (
    ID SERIAL PRIMARY KEY,
    USUARIO_ID INT UNIQUE REFERENCES USUARIO(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    CODIGO VARCHAR(50) UNIQUE NOT NULL,
    CI VARCHAR(20) UNIQUE NOT NULL,
    NOMBRES VARCHAR(100) NOT NULL,
    APELLIDO_PATERNO VARCHAR(100) NOT NULL,
    APELLIDO_MATERNO VARCHAR(100),
    FECHA_NACIMIENTO DATE,
    NACIONALIDAD VARCHAR(50),
    SEXO VARCHAR(1),
    DIRECCION TEXT,
    TELEFONO VARCHAR(20),
    EMAIL VARCHAR(150) UNIQUE NOT NULL
);

CREATE TABLE ADMINISTRACION (
    ID INT PRIMARY KEY REFERENCES PERFIL(ID) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE POSTULANTE (
    ID INT PRIMARY KEY REFERENCES PERFIL(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    COLEGIO_PROCEDENCIA VARCHAR(150),
    CIUDAD VARCHAR(100)
);

CREATE TABLE DOCENTE (
    ID INT PRIMARY KEY REFERENCES PERFIL(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    PROFESION VARCHAR(100),
    AREA_PROFESIONAL VARCHAR(100),
    GRADO_ACADEMICO VARCHAR(100),
    MAESTRIA VARCHAR(2) DEFAULT 'No', 
    DIPLOMADO_EDUCACION_SUPERIOR VARCHAR(2) DEFAULT 'No',
    EXPERIENCIA_ANOS INT,
    GRUPOS_MAXIMOS INT DEFAULT 4
);

-- 4. ESTRUCTURA ACADÉMICA Y POSTULACIÓN
CREATE TABLE FACULTAD (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(150) NOT NULL
);

CREATE TABLE GESTION (
    ID SERIAL PRIMARY KEY,
    SEMESTRE INT NOT NULL,
    AÑO INT NOT NULL
);

CREATE TABLE CARRERA (
    CODIGO SERIAL PRIMARY KEY,
    FACULTAD_ID INT REFERENCES FACULTAD(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    NOMBRE VARCHAR(150) NOT NULL,
    CUPO_MAXIMO INT NOT NULL
);

CREATE TABLE POSTULACION (
    CODIGO SERIAL PRIMARY KEY,
    POSTULANTE_ID INT REFERENCES POSTULANTE(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    GESTION_ID INT REFERENCES GESTION(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    FECHA DATE NOT NULL,
    HORA TIME NOT NULL,
    ESTADO VARCHAR(50)
);

CREATE TABLE POSTULACION_CARRERA (
    POSTULACION_CODIGO INT REFERENCES POSTULACION(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    CARRERA_CODIGO INT REFERENCES CARRERA(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    PRIORIDAD INT NOT NULL, 
    PRIMARY KEY (POSTULACION_CODIGO, CARRERA_CODIGO)
);

CREATE TABLE PAGO (
    ID SERIAL PRIMARY KEY,
    POSTULACION_CODIGO INT REFERENCES POSTULACION(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    NRO_RECIBO VARCHAR(100) UNIQUE NOT NULL,
    MONTO DECIMAL(10,2) NOT NULL,
    METODO_PAGO VARCHAR(50) NOT NULL,
    TRANSACCION_ID VARCHAR(100),
    ESTADO VARCHAR(50),
    FECHA DATE NOT NULL
);

CREATE TABLE DOCUMENTOS (
    ID SERIAL PRIMARY KEY,
    POSTULACION_CODIGO INT REFERENCES POSTULACION(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    TIPO_DOCUMENTO VARCHAR(100) NOT NULL,
    URL_ARCHIVO TEXT NOT NULL,
    ESTADO_VALIDACION VARCHAR(50) DEFAULT 'PENDIENTE'
);

-- 5. MÓDULO DE MATERIAS, GRUPOS Y HORARIOS
CREATE TABLE MATERIA (
    ID SERIAL PRIMARY KEY,
    NOMBRE VARCHAR(100) NOT NULL
);

CREATE TABLE AULA (
    ID SERIAL PRIMARY KEY,
    NRO_AULA VARCHAR(100) NOT NULL,
    CAPACIDAD INT NOT NULL
);

CREATE TABLE TURNO (
    ID SERIAL PRIMARY KEY,
    DESCRIPCION VARCHAR(50) NOT NULL 
);

CREATE TABLE GRUPO (
    CODIGO SERIAL PRIMARY KEY,
    MATERIA_ID INT REFERENCES MATERIA(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    NOMBRE VARCHAR(50) NOT NULL,
    INSCRITOS_ACTUALES INT DEFAULT 0,
    CUPO INT NOT NULL,
    MODALIDAD VARCHAR(50) 
);

CREATE TABLE HORARIO (
    ID SERIAL PRIMARY KEY,
    GRUPO_CODIGO INT REFERENCES GRUPO(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    AULA_ID INT REFERENCES AULA(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    TURNO_ID INT REFERENCES TURNO(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    DIA VARCHAR(20) NOT NULL,
    HORA_INICIO TIME NOT NULL,
    HORA_FIN TIME NOT NULL
);

CREATE TABLE CARGA_HORARIA (
    DOCENTE_ID INT REFERENCES DOCENTE(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    GRUPO_CODIGO INT REFERENCES GRUPO(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    PRIMARY KEY (DOCENTE_ID, GRUPO_CODIGO)
);

CREATE TABLE INSCRIPCIONES_CUP (
    ID SERIAL PRIMARY KEY,
    POSTULACION_CODIGO INT REFERENCES POSTULACION(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    GRUPO_CODIGO INT REFERENCES GRUPO(CODIGO) ON DELETE CASCADE ON UPDATE CASCADE,
    FECHA_INSCRIPCION DATE NOT NULL,
    ESTADO VARCHAR(50)
);

CREATE TABLE EVALUACIONES (
    ID SERIAL PRIMARY KEY,
    INSCRIPCION_ID INT REFERENCES INSCRIPCIONES_CUP(ID) ON DELETE CASCADE ON UPDATE CASCADE,
    NOTA_P1 DECIMAL(5,2) DEFAULT 0,
    NOTA_P2 DECIMAL(5,2) DEFAULT 0,
    NOTA_P3 DECIMAL(5,2) DEFAULT 0,
    PROMEDIO_FINAL DECIMAL(5,2) DEFAULT 0,
    ESTADO_MATERIA VARCHAR(50)
);
